cmake_minimum_required(VERSION 3.22)
project(oled_u8g2_host_tests LANGUAGES C)
enable_testing()
add_subdirectory(.. u8g2_port)
add_executable(test_u8g2_port test_u8g2_port.c)
target_link_libraries(test_u8g2_port PRIVATE oled_u8g2_port)
if(MSVC)
    target_compile_options(test_u8g2_port PRIVATE /W4 /WX)
    target_compile_options(oled_u8g2_port PRIVATE /Gy /Gw)
    target_link_options(test_u8g2_port PRIVATE /OPT:REF)
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_u8g2_port PRIVATE -Wall -Wextra -Werror -Wpedantic)
    target_link_options(test_u8g2_port PRIVATE -Wl,--gc-sections)
endif()
add_test(NAME u8g2_port_pixels_ownership_font COMMAND test_u8g2_port)
